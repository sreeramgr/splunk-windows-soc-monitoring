/* 1. Encoded PowerShell Execution Detected */
index=soc_lab source="*Sysmon*" "<EventID>1</EventID>" "EncodedCommand"


/* 2. Repeated Failed Windows Logins Detected */
index=soc_lab source="WinEventLog:Security" EventCode=4625 earliest=-5m@m latest=@m
| eval TargetAccount=mvindex(Account_Name,-1)
| stats count AS FailedAttempts earliest(_time) AS FirstAttempt latest(_time) AS LastAttempt values(Failure_Reason) AS FailureReason by host TargetAccount
| where FailedAttempts >= 3
| convert ctime(FirstAttempt) ctime(LastAttempt)
| sort - FailedAttempts


/* 3. Windows User Account Activity Detected */
index=soc_lab source="WinEventLog:Security" (EventCode=4720 OR EventCode=4726)
| eval SubjectUserName=mvindex(Account_Name,0)
| eval TargetUserName=mvindex(Account_Name,-1)
| eval Activity=case(
    EventCode=4720, "User account created",
    EventCode=4726, "User account deleted"
  )
| table _time host EventCode Activity SubjectUserName TargetUserName
| sort - _time


/* 4. Local Administrators Group Changed */
index=soc_lab source="WinEventLog:Security" (EventCode=4732 OR EventCode=4733) Group_Name="Administrators"
| eval Activity=case(
    EventCode=4732, "Member added to Administrators",
    EventCode=4733, "Member removed from Administrators"
)
| table _time host EventCode Activity Account_Name Member_Name Group_Name
| sort - _time


/* 5. Suspicious Windows Utility Execution */
index=soc_lab source="*Sysmon*" "<EventID>1</EventID>"
| rex field=_raw "<Data Name='Image'>(?<ProcessImage>[^<]+)"
| rex field=_raw "<Data Name='CommandLine'>(?<CommandLine>[^<]+)"
| rex field=_raw "<Data Name='ParentImage'>(?<ParentImage>[^<]+)"
| rex field=_raw "<Data Name='User'>(?<User>[^<]+)"
| eval SuspiciousReason=case(
    match(ProcessImage,"(?i)\\\\certutil\.exe$") AND match(CommandLine,"(?i)(-hashfile|-urlcache|-decode|-encode)"), "Suspicious Certutil command",
    match(ProcessImage,"(?i)\\\\mshta\.exe$"), "MSHTA execution",
    match(ProcessImage,"(?i)\\\\regsvr32\.exe$") AND match(CommandLine,"(?i)(/i:|http|scrobj)"), "Suspicious Regsvr32 command",
    match(ProcessImage,"(?i)\\\\rundll32\.exe$") AND match(CommandLine,"(?i)(javascript:|http|\\\\temp\\\\)"), "Suspicious Rundll32 command"
)
| where isnotnull(SuspiciousReason)
| table _time host User ProcessImage ParentImage CommandLine SuspiciousReason
| sort - _time


/* 6. New Windows Service Installed */
index=soc_lab source="WinEventLog:System" EventCode=7045
| table _time host EventCode Service_Name Service_File_Name Service_Type Service_Start_Type
| sort - _time
